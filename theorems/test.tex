\begin{proposition}
Suppose $-3 < \lambda < 3$. Then
\[
\langle r\rangle^\lambda \in A_2(\mathbb{R}^3).
\]
\end{proposition}

\begin{proof}
Let $B(x_0,R)$ be a ball. We consider the quantity
\[
\left(\frac{1}{|B|}\int_B w(x)\,dx\right)
\left(\frac{1}{|B|}\int_B w(x)^{-1}\,dx\right).
\]
\end{proof}
